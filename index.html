Firebase Firestore Rules (Cloud Live Sync)

Firebase Console → Firestore Database → Rules තුළ පහත code එක Paste කර Publish කරන්න

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /artifacts/{document=**} {
      allow read, write: if true;
    }

    function getUserRole() {
      return get(/databases/$(database)/documents/users/$(request.auth.token.email)).data.role;
    }

    function isAdmin() {
      return request.auth != null && (
        request.auth.token.email == "lalanadesilva@googlemail.com" ||
        request.auth.token.email == "admin@freesat.lk" ||
        getUserRole() == "admin"
      );
    }

    function isTech() {
      return request.auth != null && (
        request.auth.token.email == "tech@freesat.lk" ||
        getUserRole() == "technician"
      );
    }

    match /users/{email} {
      allow read: if request.auth != null;
      allow write: if isAdmin();
    }

    match /inventory/{itemId} {
      allow read: if request.auth != null;
      allow create, delete: if isAdmin();
      allow update: if isAdmin() || (
        isTech() &&
        request.resource.data.diff(resource.data).affectedKeys().hasOnly(['qty']) &&
        request.resource.data.qty <= resource.data.qty
      );
    }

    match /repairs/{repairId} {
      allow read: if request.auth != null;
      allow create: if isAdmin() || isTech();
      allow update, delete: if isAdmin();
    }
  }
}
